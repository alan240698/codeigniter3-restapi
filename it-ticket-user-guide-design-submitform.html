6.3. Detailed Job Implementations

Job 1: SLA Monitor (Every 5 minutes)

// app/Console/Commands/CheckSLA.php

namespace App\Console\Commands;

use Illuminate\Console\Command;
use App\Models\Ticket;
use App\Jobs\SendTicketEmail;
use Carbon\Carbon;

class CheckSLA extends Command
{
    protected $signature = 'tickets:check-sla';
    protected $description = 'Monitor SLA deadlines and send warnings';

    public function handle()
    {
        $this->info('Starting SLA check...');
        
        // Get all open tickets with SLA
        $tickets = Ticket::whereIn('status', ['new', 'in_progress', 'pending_approval'])
                        ->with(['sla', 'assignedAgent', 'assignedTeam'])
                        ->get();
        
        $warningCount = 0;
        $breachCount = 0;
        
        foreach ($tickets as $ticket) {
            // Check First Response SLA
            if (!$ticket->sla->first_response_completed_at) {
                $this->checkFirstResponseSLA($ticket, $warningCount, $breachCount);
            }
            
            // Check Resolution SLA
            if (!$ticket->sla->resolved_at) {
                $this->checkResolutionSLA($ticket, $warningCount, $breachCount);
            }
        }
        
        $this->info("SLA check completed. Warnings: {$warningCount}, Breaches: {$breachCount}");
    }
    
    private function checkFirstResponseSLA($ticket, &$warningCount, &$breachCount)
    {
        $now = Carbon::now();
        $deadline = Carbon::parse($ticket->sla->first_response_deadline);
        $totalMinutes = $ticket->sla->first_response_target_minutes;
        $remainingMinutes = $now->diffInMinutes($deadline, false);
        
        // Calculate percentage used
        $percentUsed = (($totalMinutes - $remainingMinutes) / $totalMinutes) * 100;
        
        // Send warning at 75%
        if ($percentUsed >= 75 && $percentUsed < 100 && !$ticket->sla->first_response_warning_sent) {
            $this->sendSLAWarning($ticket, 'first_response', $remainingMinutes);
            $ticket->sla->update(['first_response_warning_sent' => true]);
            $warningCount++;
        }
        
        // Send breach alert at 100%
        if ($remainingMinutes <= 0 && !$ticket->sla->first_response_breached) {
            $this->sendSLABreach($ticket, 'first_response');
            $ticket->sla->update([
                'first_response_breached' => true,
                'is_breached' => true
            ]);
            $breachCount++;
        }
    }
    
    private function checkResolutionSLA($ticket, &$warningCount, &$breachCount)
    {
        $now = Carbon::now();
        $deadline = Carbon::parse($ticket->sla->resolution_deadline);
        $totalMinutes = $ticket->sla->resolution_target_minutes;
        $remainingMinutes = $now->diffInMinutes($deadline, false);
        
        $percentUsed = (($totalMinutes - $remainingMinutes) / $totalMinutes) * 100;
        
        // Send warning at 75%
        if ($percentUsed >= 75 && $percentUsed < 100 && !$ticket->sla->resolution_warning_sent) {
            $this->sendSLAWarning($ticket, 'resolution', $remainingMinutes);
            $ticket->sla->update(['resolution_warning_sent' => true]);
            $warningCount++;
        }
        
        // Send breach alert at 100%
        if ($remainingMinutes <= 0 && !$ticket->sla->resolution_breached) {
            $this->sendSLABreach($ticket, 'resolution');
            $ticket->sla->update([
                'resolution_breached' => true,
                'is_breached' => true
            ]);
            $breachCount++;
        }
    }
    
    private function sendSLAWarning($ticket, $type, $remainingMinutes)
    {
        $recipients = [
            ['email' => $ticket->assignedAgent->email, 'name' => $ticket->assignedAgent->name],
            ['email' => $ticket->assignedTeam->lead->email, 'name' => $ticket->assignedTeam->lead->name]
        ];
        
        SendTicketEmail::dispatch($ticket, 'sla_warning', $recipients, [
            'sla_type' => $type,
            'time_remaining' => $this->formatMinutes($remainingMinutes)
        ]);
        
        $this->info("SLA Warning sent for ticket {$ticket->ticket_id}");
    }
    
    private function sendSLABreach($ticket, $type)
    {
        // Get management chain
        $managementChain = $this->getManagementChain($ticket->assignedTeam);
        
        SendTicketEmail::dispatch($ticket, 'sla_breach', $managementChain, [
            'sla_type' => $type
        ]);
        
        // Log breach incident
        \Log::critical("SLA BREACH: Ticket {$ticket->ticket_id} - {$type} SLA breached");
        
        $this->error("SLA BREACH for ticket {$ticket->ticket_id}");
    }
    
    private function formatMinutes($minutes)
    {
        if ($minutes < 60) {
            return "{$minutes} minutes";
        } else if ($minutes < 1440) {
            $hours = round($minutes / 60, 1);
            return "{$hours} hours";
        } else {
            $days = round($minutes / 1440, 1);
            return "{$days} days";
        }
    }
    
    private function getManagementChain($team)
    {
        return [
            ['email' => $team->lead->email, 'name' => $team->lead->name],
            ['email' => $team->manager->email, 'name' => $team->manager->name],
            ['email' => config('tickets.director_email'), 'name' => 'Director']
        ];
    }
}

Job 2: Auto Escalation (Every 10 minutes)

// app/Console/Commands/AutoEscalate.php

namespace App\Console\Commands;

use Illuminate\Console\Command;
use App\Models\Ticket;
use App\Jobs\SendTicketEmail;
use Carbon\Carbon;

class AutoEscalate extends Command
{
    protected $signature = 'tickets:auto-escalate';
    protected $description = 'Auto-escalate tickets exceeding time limits';

    public function handle()
    {
        $this->info('Checking for tickets to escalate...');
        
        $tickets = Ticket::whereIn('status', ['in_progress', 'pending_approval'])
                        ->with(['escalation', 'assignedTeam', 'assignedAgent'])
                        ->get();
        
        $escalatedCount = 0;
        
        foreach ($tickets as $ticket) {
            if ($this->shouldEscalate($ticket)) {
                $this->escalateTicket($ticket);
                $escalatedCount++;
            }
        }
        
        $this->info("Escalation check completed. Escalated: {$escalatedCount} tickets");
    }
    
    private function shouldEscalate($ticket)
    {
        if (!$ticket->escalation || !$ticket->escalation->enabled) {
            return false;
        }
        
        $currentLevel = $ticket->current_level;
        $lastLevelChange = Carbon::parse($ticket->last_level_change);
        $timeInLevel = $lastLevelChange->diffInHours(now());
        
        $escalationRule = $ticket->escalation->levels
            ->where('level', $currentLevel)
            ->first();
        
        if (!$escalationRule) {
            return false;
        }
        
        return $timeInLevel >= $escalationRule->time_limit_hours;
    }
    
    private function escalateTicket($ticket)
    {
        $currentLevel = $ticket->current_level;
        $nextLevel = $currentLevel + 1;
        
        $nextLevelRule = $ticket->escalation->levels
            ->where('level', $nextLevel)
            ->first();
        
        if (!$nextLevelRule) {
            $this->warn("No next level found for ticket {$ticket->ticket_id}");
            return;
        }
        
        $oldTeam = $ticket->assignedTeam;
        $newTeam = \App\Models\Team::find($nextLevelRule->team_id);
        
        // Update ticket
        $ticket->update([
            'assigned_team_id' => $newTeam->id,
            'assigned_agent_id' => null, // Unassign from specific agent
            'current_level' => $nextLevel,
            'last_level_change' => now(),
            'escalated_at' => now(),
            'escalation_count' => $ticket->escalation_count + 1
        ]);
        
        // Log activity
        $ticket->activities()->create([
            'action' => 'escalated',
            'old_value' => json_encode(['team' => $oldTeam->name, 'level' => $currentLevel]),
            'new_value' => json_encode(['team' => $newTeam->name, 'level' => $nextLevel]),
            'comment' => "Auto-escalated due to time limit exceeded"
        ]);
        
        // Send notifications
        $this->sendEscalationEmails($ticket, $oldTeam, $newTeam);
        
        $this->info("Escalated ticket {$ticket->ticket_id} from L{$currentLevel} to L{$nextLevel}");
        
        // Log critical escalation
        \Log::warning("ESCALATION: Ticket {$ticket->ticket_id} escalated to Level {$nextLevel}");
    }
    
    private function sendEscalationEmails($ticket, $oldTeam, $newTeam)
    {
        // Notify old team
        SendTicketEmail::dispatch($ticket, 'ticket_escalated', [
            ['email' => $oldTeam->lead->email, 'name' => $oldTeam->lead->name]
        ], [
            'escalation_reason' => 'Time limit exceeded',
            'old_team' => $oldTeam->name,
            'new_team' => $newTeam->name
        ]);
        
        // Notify new team
        SendTicketEmail::dispatch($ticket, 'ticket_assigned', 
            $newTeam->members->map(fn($m) => ['email' => $m->email, 'name' => $m->name])->toArray(),
            ['escalated' => true]
        );
        
        // Notify management
        SendTicketEmail::dispatch($ticket, 'ticket_escalated', [
            ['email' => $newTeam->manager->email, 'name' => $newTeam->manager->name]
        ], [
            'escalation_reason' => 'Time limit exceeded',
            'escalation_level' => $ticket->current_level
        ]);
    }
}

Job 3: Auto-close Stale Tickets (Daily)

// app/Console/Commands/AutoCloseStale.php

namespace App\Console\Commands;

use Illuminate\Console\Command;
use App\Models\Ticket;
use App\Jobs\SendTicketEmail;
use Carbon\Carbon;

class AutoCloseStale extends Command
{
    protected $signature = 'tickets:auto-close-stale';
    protected $description = 'Auto-close tickets with no response for 7 days';

    public function handle()
    {
        $this->info('Checking for stale tickets...');
        
        $sevenDaysAgo = Carbon::now()->subDays(7);
        
        // Find tickets in "On Hold" or "Resolved" status with no activity for 7 days
        $staleTickets = Ticket::where(function($query) {
                $query->where('status', 'on_hold')
                      ->orWhere('status', 'resolved');
            })
            ->where('updated_at', '<', $sevenDaysAgo)
            ->with('user')
            ->get();
        
        $closedCount = 0;
        
        foreach ($staleTickets as $ticket) {
            // Check if there's any recent activity
            $lastActivity = $ticket->activities()
                                  ->latest()
                                  ->first();
            
            if ($lastActivity && Carbon::parse($lastActivity->created_at)->gt($sevenDaysAgo)) {
                continue; // Skip if there's recent activity
            }
            
            // Auto-close ticket
            $ticket->update([
                'status' => 'closed',
                'closed_at' => now(),
                'closed_by' => 'system',
                'close_reason' => 'Auto-closed due to inactivity'
            ]);
            
            // Log activity
            $ticket->activities()->create([
                'action' => 'auto_closed',
                'comment' => 'Ticket auto-closed after 7 days of inactivity'
            ]);
            
            // Send notification to user
            SendTicketEmail::dispatch($ticket, 'ticket_closed', [
                ['email' => $ticket->user->email, 'name' => $ticket->user->name]
            ], [
                'close_reason' => 'Auto-closed due to inactivity',
                'auto_closed' => true
            ]);
            
            $closedCount++;
            $this->info("Auto-closed ticket {$ticket->ticket_id}");
        }
        
        $this->info("Auto-close completed. Closed {$closedCount} stale tickets");
    }
}

Job 4: Failed Email Retry (Hourly)

// app/Console/Commands/RetryFailedEmails.php

namespace App\Console\Commands;

use Illuminate\Console\Command;
use App\Models\EmailLog;
use App\Models\Ticket;
use App\Jobs\SendTicketEmail;

class RetryFailedEmails extends Command
{
    protected $signature = 'emails:retry-failed';
    protected $description = 'Retry sending failed emails';

    public function handle()
    {
        $this->info('Checking for failed emails...');
        
        // Get failed emails with retry count < 3
        $failedEmails = EmailLog::where('status', 'failed')
                               ->where('retry_count', '<', 3)
                               ->where('created_at', '>', now()->subDays(3)) // Only retry within 3 days
                               ->get();
        
        $retriedCount = 0;
        
        foreach ($failedEmails as $log) {
            try {
                $ticket = Ticket::find($log->ticket_id);
                
                if (!$ticket) {
                    $this->warn("Ticket not found for email log {$log->id}");
                    $log->update(['status' => 'cancelled']);
                    continue;
                }
                
                $recipients = json_decode($log->recipients, true);
                $metadata = json_decode($log->metadata, true);
                
                // Retry sending
                SendTicketEmail::dispatch($ticket, $log->email_type, $recipients, $metadata);
                
                // Update retry count
                $log->increment('retry_count');
                $log->update([
                    'last_retry_at' => now()
                ]);
                
                $retriedCount++;
                $this->info("Retried email log {$log->id} (attempt {$log->retry_count})");
                
            } catch (\Exception $e) {
                $this->error("Failed to retry email log {$log->id}: {$e->getMessage()}");
                
                $log->update([
                    'error_message' => $e->getMessage(),
                    'last_retry_at' => now()
                ]);
                
                // If max retries reached, mark as permanently failed
                if ($log->retry_count >= 2) {
                    $log->update(['status' => 'permanently_failed']);
                    \Log::error("Email permanently failed after 3 attempts: {$log->id}");
                }
            }
        }
        
        $this->info("Retry completed. Retried {$retriedCount} emails");
    }
}

2.4. Email Queue Implementation

// Laravel Queue Job Example
namespace App\Jobs;

use Illuminate\Bus\Queueable;
use Illuminate\Queue\SerializesModels;
use Illuminate\Queue\InteractsWithQueue;
use Illuminate\Contracts\Queue\ShouldQueue;

class SendTicketEmail implements ShouldQueue
{
    use InteractsWithQueue, Queueable, SerializesModels;

    public $ticket;
    public $emailType;
    public $recipients;

    public function __construct($ticket, $emailType, $recipients)
    {
        $this->ticket = $ticket;
        $this->emailType = $emailType;
        $this->recipients = $recipients;
    }

    public function handle()
    {
        $template = $this->getTemplate($this->emailType);
        $data = $this->prepareEmailData($this->ticket);

        foreach ($this->recipients as $recipient) {
            Mail::to($recipient['email'])->send(
                new TicketNotification($template, $data, $recipient)
            );
        }

        // Log email activity
        EmailLog::create([
            'ticket_id' => $this->ticket->id,
            'email_type' => $this->emailType,
            'recipients' => json_encode($this->recipients),
            'sent_at' => now()
        ]);
    }

    private function getTemplate($emailType)
    {
        $templates = [
            'ticket_created' => 'emails.ticket_created',
            'approval_required' => 'emails.approval_required',
            'ticket_approved' => 'emails.ticket_approved',
            'ticket_rejected' => 'emails.ticket_rejected',
            'ticket_assigned' => 'emails.ticket_assigned',
            'status_changed' => 'emails.status_changed',
            'ticket_escalated' => 'emails.ticket_escalated',
            'sla_warning' => 'emails.sla_warning',
            'sla_breach' => 'emails.sla_breach',
            'ticket_resolved' => 'emails.ticket_resolved',
            'ticket_closed' => 'emails.ticket_closed',
            'comment_added' => 'emails.comment_added',
        ];

        return $templates[$emailType];
    }

    private function prepareEmailData($ticket)
    {
        return [
            'ticket_id' => $ticket->ticket_id,
            'ticket_title' => $ticket->title,
            'ticket_url' => url("/tickets/{$ticket->id}"),
            'user_name' => $ticket->user->name,
            'priority' => $ticket->priority,
            'assigned_team' => $ticket->assignedTeam->name,
            'sla_deadline' => $ticket->sla->resolution_deadline->format('M d, Y h:i A'),
            'first_response_time' => $ticket->sla->first_response_target,
            'resolution_time' => $ticket->sla->resolution_target,
            // Add more fields as needed
        ];
    }
}

// Example: Trigger emails when ticket is created
public function createTicket($request)
{
    // Submit to ticket system API
    $response = Http::post('api/tickets/submit', $request->all());
    $ticketData = $response->json();
    
    // Save ticket to database
    $ticket = Ticket::create([
        'ticket_id' => $ticketData['ticket_id'],
        'user_id' => auth()->id(),
        'status' => $ticketData['status'],
        // ... other fields
    ]);
    
    // Trigger emails
    $this->sendTicketCreatedEmails($ticket, $ticketData);
    
    if ($ticketData['approval']['required']) {
        $this->sendApprovalEmails($ticket, $ticketData);
    }
    
    return $ticket;
}

private function sendTicketCreatedEmails($ticket, $ticketData)
{
    // Email to ticket creator
    SendTicketEmail::dispatch(
        $ticket,
        'ticket_created',
        [['email' => $ticket->user->email, 'name' => $ticket->user->name]]
    );
    
    // Email to assigned team
    $teamMembers = Team::find($ticketData['routing']['assigned_to']['id'])
                      ->members;
    
    $recipients = $teamMembers->map(function($member) {
        return ['email' => $member->email, 'name' => $member->name];
    })->toArray();
    
    SendTicketEmail::dispatch($ticket, 'ticket_assigned', $recipients);
}

private function sendApprovalEmails($ticket, $ticketData)
{
    foreach ($ticketData['approval']['approvers'] as $approver) {
        SendTicketEmail::dispatch(
            $ticket,
            'approval_required',
            [['email' => $approver['email'], 'name' => $approver['name']]]
        );
    }
}

// Example: SLA Warning Check (Run by Scheduler)
public function checkSLAWarnings()
{
    $tickets = Ticket::where('status', '!=', 'closed')
                    ->with(['sla', 'assignedAgent'])
                    ->get();
    
    foreach ($tickets as $ticket) {
        $timeRemaining = now()->diffInMinutes($ticket->sla->resolution_deadline);
        $totalTime = $ticket->sla->resolution_target_minutes;
        $percentUsed = (($totalTime - $timeRemaining) / $totalTime) * 100;
        
        // Send warning at 75% threshold
        if ($percentUsed >= 75 && $percentUsed < 90 && !$ticket->warning_sent) {
            SendTicketEmail::dispatch(
                $ticket,
                'sla_warning',
                [
                    ['email' => $ticket->assignedAgent->email, 'name' => $ticket->assignedAgent->name],
                    ['email' => $ticket->assignedAgent->team_lead->email, 'name' => $ticket->assignedAgent->team_lead->name]
                ]
            );
            
            $ticket->update(['warning_sent' => true]);
        }
        
        // Send breach alert at 100%
        if ($timeRemaining <= 0 && !$ticket->breach_sent) {
            $managers = $this->getManagementChain($ticket->assignedTeam);
            SendTicketEmail::dispatch($ticket, 'sla_breach', $managers);
            
            $ticket->update(['breach_sent' => true]);
        }
    }
}

// Email Statistics Dashboard
public function getEmailStats()
{
    return [
        'sent_today' => EmailLog::whereDate('sent_at', today())->count(),
        'failed_today' => EmailLog::whereDate('sent_at', today())
                                 ->where('status', 'failed')->count(),
        'queue_size' => Queue::size('emails'),
        'avg_delivery_time' => EmailLog::whereDate('sent_at', today())
                                      ->avg('delivery_time_seconds'),
        'by_type' => EmailLog::whereDate('sent_at', today())
                            ->groupBy('email_type')
                            ->selectRaw('email_type, count(*) as count')
                            ->get(),
    ];
}

// Failed Email Retry
public function retryFailedEmails()
{
    $failedEmails = EmailLog::where('status', 'failed')
                           ->where('retry_count', '<', 3)
                           ->get();
    
    foreach ($failedEmails as $log) {
        SendTicketEmail::dispatch(
            Ticket::find($log->ticket_id),
            $log->email_type,
            json_decode($log->recipients, true)
        );
        
        $log->increment('retry_count');
    }
}
